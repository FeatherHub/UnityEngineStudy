public abstract class Weapon

// 템플릿 메서드: virtual이 아니라서 순서는 누구도 못 바꿈
public void TryShoot(bool pressedThisFrame, bool held)
{
    if (!WantsToFire(pressedThisFrame, held)) return;   // 훅
    if (Time.time < nextFireTime) return;               // 고정

    nextFireTime = Time.time + weaponSO.FireRate;       // 고정
    muzzleFlash.Play();                                 // 고정
    Fire();                                             // 훅
    OnAfterFire();                                      // 훅
}

// 기본은 단발. 연사 무기만 override
protected virtual bool WantsToFire(bool pressed, bool held) => pressed;

// 기본은 레이캐스트 한 발. 산탄총만 override
protected virtual void Fire() { /* Raycast + 피격 처리 */ }

// 기본은 아무것도 안 함
protected virtual void OnAfterFire() { }
