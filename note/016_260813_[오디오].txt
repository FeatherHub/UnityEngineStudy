PCM = Pulse Code Modulation

아날로그 파형

      /\
     /  \      /\
____/    \____/  \____

시간별 샘플링

      ●
   ●     ●      ●
 ●         ●  ●
●            ●

---

숫자로 저장

Sample 0 =  1203
Sample 1 =  1348
Sample 2 =  1570
Sample 3 =  1492
...

---

stereo, 44.1 kHz, 16-bit PCM이면 대략

44,100 samples/s
× 2 byte
× 2 channels

≈ 176 KB/s
≈ 10.6 MB/min

---

ADPCM = Adaptive Differential Pulse Code Modulation

Adaptive
= 상황에 맞춰 예측 규칙을 조절

Differential
= 실제 값 대신 앞 sample과의 차이를 중심으로 저장

Pulse Code Modulation
= PCM 계열의 샘플 표현

---

ADPCM은 파형이 연속적이라는 성질을 이용합니다.

첫 값 1000

이후 변화량
+12
+8
-3
+13

---

실제 ADPCM은 단순 delta 저장보다 한 단계 더 나갑니다

이전 sample들을 이용해 다음 값을 예측하고
작은 prediction error를 압축해서 저장합니다

---

Adaptive

잔잔한 구간
1000 1002 1001 1003
→ 작은 step

급격한 구간
1000 1800 700 2000
→ 큰 step

---

PCM
→ 아주 짧고 자주 재생되는 소리
→ CPU 비용 최소화가 중요할 때

ADPCM
→ 짧거나 중간 길이의 SFX
→ PCM 메모리는 부담되고
   Vorbis/MP3 디코딩 비용도 피하고 싶을 때

// 에셋을 소유하지 않는다. 빌려주고, 세팅하고, 회수한다.
public void Play(SoundEvent evt)
{
    if (!evt.TryConsumeCooldown(Time.unscaledTime)) return;

    var voice = _pool.Rent();          // 없으면 priority 낮은 것을 steal
    evt.ApplyTo(voice);                // resource, group, volume, priority
    voice.Play();
    _pool.ReturnWhenFinished(voice);
}

// ── SoundBank (ScriptableObject) ──
// 로드 단위. Addressables Group 과 1:1. 자신은 Boot 그룹에 상주하며
// AssetReference(GUID 문자열)만 들고 있어 클립을 끌고 오지 않는다.
[CreateAssetMenu(menuName = "Audio/Sound Bank")]
public class SoundBank : ScriptableObject
{
    [SerializeField] string       _bankId;            // "GameplayCore" — 생성 코드의 네임스페이스
    [SerializeField] string       _addressableGroup;  // "Audio_GameplayCore" — 검증용
    [SerializeField] SoundEvent[] _events;

    Dictionary<string, SoundEvent> _lookup;           // 최초 접근 시 1회 구축

    public string BankId => _bankId;

    // ── SoundBank::Find ──
    public SoundEvent Find(string key)
    {
        _lookup ??= _events.ToDictionary(e => e.Key);
        return _lookup.TryGetValue(key, out var e) ? e : null;
    }
}

namespace Game.Audio
{
    // 사운드 접근 — 뱅크가 곧 네임스페이스가 된다
    public static class Audio
    {
        public static class Boot
        {
            static SoundEvent _error;
            public static SoundEvent Error => _error ??= Banks.Boot.Find("Error");
        }

        public static class GameplayCore
        {
            static SoundEvent _pop, _swap, _bomb;
            public static SoundEvent Pop  => _pop  ??= Banks.GameplayCore.Find("Pop");
            public static SoundEvent Swap => _swap ??= Banks.GameplayCore.Find("Swap");
            public static SoundEvent Bomb => _bomb ??= Banks.GameplayCore.Find("Bomb");
        }
    }

    // 뱅크 접근 — 로드/해제용
    public static class Banks
    {
        public static SoundBank Boot         => BankRegistry.Get("Boot");
        public static SoundBank MainMenu     => BankRegistry.Get("MainMenu");
        public static SoundBank GameplayCore => BankRegistry.Get("GameplayCore");

        // 파라미터화된 뱅크는 팩토리로 (테마가 늘어나도 재생성 불필요)
        public static SoundBank Theme(string themeId) => BankRegistry.Get("Theme_" + themeId);
    }
}

개별 예외를 보존하는 열쇠 — importSettingsMissing

최초 import일 때만 true
최초 import 에만 규칙을 적용한다. 이후 사람이 손댄 설정은 건드리지 않는다

1. Preset Manager 기본값        프로젝트 전역 fallback
2. 부모 폴더의 .preset          상속
3. 같은 폴더의 .preset          폴더 단위 규칙
4. 파일명 prefix 규칙           이름 단위 규칙 (preset 없을 때)
5. 개별 수동 조정               파일 단위 예외  ← escape hatch. .meta 에 영구 보존

파일 단위 override
.meta 파일이 곧 파일 단위 override 저장소

최초 import   .meta 없음  →  importSettingsMissing == true   →  규칙 적용  →  결과가 .meta 에 기록
이후 import   .meta 있음  →  importSettingsMissing == false  →  규칙 skip  →  .meta 값 그대로
사람이 수정   인스펙터에서 값 변경 → Apply → .meta 갱신 → 영구 보존

If multiple classes are defined in your script, Unity selects the class with the same name as the file

// ══════════════════════════════════════════════════════════════
//  AudioLibrary.cs  —  게임의 사운드 카탈로그. 이 파일 하나가 전부
// ══════════════════════════════════════════════════════════════
namespace Game.Audio
{
    [CreateAssetMenu(menuName = "Audio/Library")]
    public class AudioLibrary : ScriptableObject
    {
        public GameSounds  Game;
        public UISounds    UI;
        public MusicSounds Music;

        // ─── 카테고리 = 로드 단위 ─── 필드 한 줄 = 사운드 하나 등록
        [Serializable]
        public class GameSounds : SoundCategory
        {
            public SoundEventRef Win, Pop, Bomb, Cascade, SwapFail;
        }

        [Serializable]
        public class UISounds : SoundCategory
        {
            public SoundEventRef Click, PopupOpen, PopupClose, Error;
        }

        [Serializable]
        public class MusicSounds : SoundCategory
        {
            public SoundEventRef Lobby, Stage, Result;
        }
    }

    // ─── 호출부가 보는 얼굴 ───
    public static class Audio
    {
        static AudioLibrary L => AudioRuntime.Library;   // 부트에서 1회 주입

        public static AudioLibrary.GameSounds  Game  => L.Game;
        public static AudioLibrary.UISounds    UI    => L.UI;
        public static AudioLibrary.MusicSounds Music => L.Music;
    }
}
